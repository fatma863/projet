\chapter{Annexe B — Extrait de la spécification OpenAPI}

La documentation complète de l'API REST est disponible à l'URL
\texttt{/api/documentation} (Swagger UI) après déploiement du backend.
L'extrait ci-dessous illustre les endpoints principaux.

\begin{lstlisting}[style=yamlstyle,caption={OpenAPI 3.0 — Endpoints d'authentification et de projets},label={lst:openapi}]
openapi: "3.0.3"
info:
  title: TAC-TIC IntelliPM API
  version: "1.0.0"
  description: API REST pour la plateforme de gestion de projets

paths:
  /api/auth/login:
    post:
      summary: Connexion utilisateur
      tags: [Auth]
      requestBody:
        content:
          application/json:
            schema:
              type: object
              required: [email, password]
              properties:
                email:    { type: string, format: email }
                password: { type: string, minLength: 8 }
      responses:
        "200":
          description: Connexion réussie
          content:
            application/json:
              schema:
                type: object
                properties:
                  token: { type: string }
                  user:  { $ref: "#/components/schemas/User" }
        "422":
          description: Identifiants invalides ou compte désactivé

  /api/projects:
    get:
      summary: Liste des projets de l'utilisateur
      tags: [Projets]
      security: [{ bearerAuth: [] }]
      parameters:
        - name: status
          in: query
          schema:
            type: string
            enum: [planning, active, suspended, archived]
      responses:
        "200":
          description: Liste paginée des projets
    post:
      summary: Créer un projet
      tags: [Projets]
      security: [{ bearerAuth: [] }]
      requestBody:
        content:
          application/json:
            schema:
              type: object
              required: [name, code]
              properties:
                name:       { type: string }
                code:       { type: string, maxLength: 10 }
                priority:   { type: string, enum: [low, medium, high, urgent] }
                start_date: { type: string, format: date }
                end_date:   { type: string, format: date }

  /api/ml/recommend/{taskId}:
    get:
      summary: Recommandation ML d'assignation
      tags: [IA / ML]
      security: [{ bearerAuth: [] }]
      parameters:
        - name: taskId
          in: path
          required: true
          schema: { type: integer }
      responses:
        "200":
          description: Liste de candidats classés par score
          content:
            application/json:
              schema:
                type: array
                items:
                  type: object
                  properties:
                    user_id:     { type: integer }
                    name:        { type: string }
                    score:       { type: number, format: float }
                    explanation: { type: string }
        "503":
          description: Microservice ML indisponible

components:
  securitySchemes:
    bearerAuth:
      type: http
      scheme: bearer
      bearerFormat: JWT
  schemas:
    User:
      type: object
      properties:
        id:     { type: integer }
        name:   { type: string }
        email:  { type: string }
        role:   { type: string, enum: [admin, member, client] }
        avatar: { type: string, nullable: true }
\end{lstlisting}
